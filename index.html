<script>

    // The "Brain" for calculations

    function calculate() {

        const d1 = parseFloat(document.getElementById('dist1').value);

        const t1 = parseFloat(document.getElementById('time1').value);

        const d2 = parseFloat(document.getElementById('dist2').value);


        if (d1 > 0 && t1 > 0 && d2 > 0) {

            const t2 = t1 * Math.pow((d2 / d1), 1.06);

            const hours = Math.floor(t2 / 60);

            const minutes = Math.floor(t2 % 60);

            document.getElementById('predTime').innerText = `${hours}h ${minutes}m`;


            const paceDec = t2 / d2;

            const paceMin = Math.floor(paceDec);

            const paceSec = Math.round((paceDec - paceMin) * 60);

            document.getElementById('targetPace').innerText = `${paceMin}:${paceSec < 10 ? '0' : ''}${paceSec}`;


            const easy = paceDec * 1.25;

            document.getElementById('easyPace').innerText = `${Math.floor(easy)}:${Math.round((easy % 1) * 60)} min/mile`;

            const tempo = paceDec * 1.05;

            document.getElementById('tempoPace').innerText = `${Math.floor(tempo)}:${Math.round((tempo % 1) * 60)} min/mile`;

        }

    }


    // The Reset Function

    function resetForm() {

        document.getElementById('dist1').value = 3.1;

        document.getElementById('time1').value = 25;

        document.getElementById('dist2').value = 26.2;

        calculate(); // Run calculation once more to refresh the display

    }


    document.querySelectorAll('input').forEach(input => {

        input.addEventListener('input', calculate);

    });


    calculate();

</script>
